find_program(GLIB_COMPILE_RESOURCES NAMES glib-compile-resources REQUIRED)

set(XOURNALPP_GRESOURCE_C   ${CMAKE_CURRENT_BINARY_DIR}/xournalpp.gresource.c)
set(XOURNALPP_GRESOURCE_C   ${XOURNALPP_GRESOURCE_C} PARENT_SCOPE)
set(XOURNALPP_GRESOURCE_XML ${CMAKE_CURRENT_BINARY_DIR}/xournalpp.gresource.xml)

list(JOIN XOURNALPP_GRESOURCES_FILES "</file>\n<file>" XOURNALPP_GRESOURCE_XML_CONTENT)
string(PREPEND XOURNALPP_GRESOURCE_XML_CONTENT "<?xml version=\"1.0\" encoding=\"UTF-8\"?><gresources><gresource prefix=\"" ${XOURNALPP_GRESOURCE_NAMESPACE} "\">\n<file>")

foreach(resourcefile ${GENERATED_INI_FILES})
    cmake_path(GET resourcefile FILENAME filename)
    string(APPEND XOURNALPP_GRESOURCE_XML_CONTENT "</file>\n<file alias=\"" ${filename} "\">" ${resourcefile})
endforeach()

string(APPEND XOURNALPP_GRESOURCE_XML_CONTENT "</file>\n</gresource></gresources>")

file(WRITE ${XOURNALPP_GRESOURCE_XML} ${XOURNALPP_GRESOURCE_XML_CONTENT})

add_custom_command(
    OUTPUT ${XOURNALPP_GRESOURCE_C}
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    COMMAND ${GLIB_COMPILE_RESOURCES}
    ARGS
        --generate-source
        --sourcedir=${CMAKE_CURRENT_SOURCE_DIR}
        --target=${XOURNALPP_GRESOURCE_C}
        ${XOURNALPP_GRESOURCE_XML}
    VERBATIM
    MAIN_DEPENDENCY ${XOURNALPP_GRESOURCE_XML}
    DEPENDS
        ${XOURNALPP_GRESOURCES_FILES}
        ${GENERATED_INI_FILES}
)

add_custom_target(
    xournalpp-resource
    DEPENDS ${XOURNALPP_GRESOURCE_C}
)
